cmake_minimum_required(VERSION 3.20)
project(Task_C_v1_profile LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(MPI REQUIRED COMPONENTS CXX)
find_package(OpenMP REQUIRED COMPONENTS CXX)
add_library(cg_common INTERFACE)
target_include_directories(cg_common INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include )
target_compile_features(cg_common INTERFACE cxx_std_17)
add_library(cg_ops STATIC src/common/csr_ops.cpp)
target_link_libraries(cg_ops PUBLIC cg_common)
add_library(cg_poisson STATIC src/common/poisson.cpp)
target_link_libraries(cg_poisson PUBLIC cg_common)
add_library(cg_matrix_market STATIC src/common/matrix_market.cpp)
target_link_libraries(cg_matrix_market PUBLIC cg_common)
add_library(cg_mpi_ops STATIC src/mpi/mpi_partition.cpp src/mpi/mpi_ops.cpp)
target_link_libraries(cg_mpi_ops PUBLIC cg_common MPI::MPI_CXX)
add_library(cg_hybrid_ops STATIC src/hybrid/hybrid_ops.cpp src/hybrid/hybrid_workspace.cpp)
target_link_libraries(cg_hybrid_ops PUBLIC cg_mpi_ops OpenMP::OpenMP_CXX)
add_library(cg_hybrid STATIC src/hybrid/hybrid_solver.cpp)
target_link_libraries(cg_hybrid PUBLIC cg_hybrid_ops)
add_library(cg_hybrid_benchmark STATIC src/hybrid/benchmark_hybrid.cpp src/hybrid/benchmark_hybrid_csv.cpp)
target_link_libraries(cg_hybrid_benchmark PUBLIC cg_hybrid)
add_executable(run_hybrid_benchmark src/hybrid/run_hybrid_benchmark.cpp)
target_link_libraries(run_hybrid_benchmark PRIVATE cg_hybrid_benchmark cg_poisson cg_matrix_market)

# Original profiled comparators from the same Member C revision.
add_library(cg_mpi STATIC src/mpi/mpi_solver.cpp)
target_link_libraries(cg_mpi PUBLIC cg_mpi_ops)
add_library(cg_mpi_benchmark STATIC src/mpi/benchmark_mpi.cpp src/mpi/benchmark_mpi_csv.cpp)
target_link_libraries(cg_mpi_benchmark PUBLIC cg_mpi)
add_executable(run_mpi_benchmark src/mpi/run_mpi_benchmark.cpp)
target_link_libraries(run_mpi_benchmark PRIVATE cg_mpi_benchmark cg_poisson cg_matrix_market)
add_library(cg_openmp_ops STATIC src/openmp/openmp_ops.cpp)
target_link_libraries(cg_openmp_ops PUBLIC cg_common OpenMP::OpenMP_CXX)
add_library(cg_openmp STATIC src/openmp/cg_openmp.cpp)
target_link_libraries(cg_openmp PUBLIC cg_openmp_ops)
add_library(cg_openmp_benchmark STATIC src/openmp/benchmark_openmp.cpp src/openmp/benchmark_openmp_csv.cpp)
target_link_libraries(cg_openmp_benchmark PUBLIC cg_openmp)
add_executable(run_openmp_benchmark src/openmp/run_openmp_benchmark.cpp)
target_link_libraries(run_openmp_benchmark PRIVATE cg_openmp_benchmark cg_poisson cg_matrix_market cg_ops)

include(CTest)
if(BUILD_TESTING)
    add_executable(test_hybrid_ops tests/test_hybrid_ops.cpp)
    target_link_libraries(test_hybrid_ops PRIVATE cg_hybrid_ops cg_mpi_ops cg_ops cg_poisson)
    add_executable(test_hybrid_cg tests/test_hybrid_cg.cpp)
    target_link_libraries(test_hybrid_cg PRIVATE cg_hybrid cg_poisson)
    foreach(target test_hybrid_ops test_hybrid_cg)
        target_compile_options(${target} PRIVATE -UNDEBUG)
    endforeach()
    add_test(NAME hybrid_ops COMMAND ${MPIEXEC_EXECUTABLE} ${MPIEXEC_NUMPROC_FLAG} 2
        ${MPIEXEC_PREFLAGS} $<TARGET_FILE:test_hybrid_ops> ${MPIEXEC_POSTFLAGS})
    set_tests_properties(hybrid_ops PROPERTIES TIMEOUT 90 PROCESSORS 4)
    foreach(ranks 2 4)
        add_test(NAME hybrid_cg_${ranks} COMMAND ${MPIEXEC_EXECUTABLE} ${MPIEXEC_NUMPROC_FLAG} ${ranks}
            ${MPIEXEC_PREFLAGS} $<TARGET_FILE:test_hybrid_cg> ${MPIEXEC_POSTFLAGS})
        math(EXPR workers "${ranks} * 2")
        set_tests_properties(hybrid_cg_${ranks} PROPERTIES TIMEOUT 90 PROCESSORS ${workers})
    endforeach()
endif()
