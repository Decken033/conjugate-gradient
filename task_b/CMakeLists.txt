cmake_minimum_required(VERSION 3.20)
project(DSA5208_Task_B LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(MPI REQUIRED COMPONENTS CXX)

# Original production target source lists and link dependencies.
add_library(cg_common INTERFACE)
target_include_directories(cg_common INTERFACE ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(cg_common INTERFACE cxx_std_17)
add_library(cg_poisson STATIC src/common/poisson.cpp)
target_link_libraries(cg_poisson PUBLIC cg_common)
add_library(cg_matrix_market STATIC src/common/matrix_market.cpp)
target_link_libraries(cg_matrix_market PUBLIC cg_common)
add_library(cg_mpi_ops STATIC src/mpi/mpi_partition.cpp src/mpi/mpi_ops.cpp)
target_link_libraries(cg_mpi_ops PUBLIC cg_common MPI::MPI_CXX)
add_library(cg_mpi STATIC src/mpi/mpi_solver.cpp)
target_link_libraries(cg_mpi PUBLIC cg_mpi_ops)
add_library(cg_mpi_benchmark STATIC src/mpi/benchmark_mpi.cpp src/mpi/benchmark_mpi_csv.cpp)
target_link_libraries(cg_mpi_benchmark PUBLIC cg_mpi)
add_executable(run_mpi_benchmark src/mpi/run_mpi_benchmark.cpp)
target_link_libraries(run_mpi_benchmark PRIVATE cg_mpi_benchmark cg_poisson cg_matrix_market)

include(CTest)
if(BUILD_TESTING)
    # The serial reference is test-only and is local to this package.
    add_library(cg_ops STATIC src/common/csr_ops.cpp)
    target_link_libraries(cg_ops PUBLIC cg_common)
    add_library(cg_serial STATIC src/serial/cg_serial.cpp)
    target_link_libraries(cg_serial PUBLIC cg_ops)
    add_executable(test_mpi_partition tests/test_mpi_partition.cpp)
    target_link_libraries(test_mpi_partition PRIVATE cg_mpi_ops cg_poisson)
    add_executable(test_mpi_spmv tests/test_mpi_spmv.cpp)
    target_link_libraries(test_mpi_spmv PRIVATE cg_mpi_ops cg_ops cg_poisson)
    add_executable(test_mpi_cg tests/test_mpi_cg.cpp)
    target_link_libraries(test_mpi_cg PRIVATE cg_mpi cg_serial cg_poisson)
    add_executable(test_mpi_benchmark tests/test_mpi_benchmark.cpp)
    target_link_libraries(test_mpi_benchmark PRIVATE cg_mpi_benchmark cg_poisson)
    foreach(target test_mpi_partition test_mpi_spmv test_mpi_cg test_mpi_benchmark)
        target_compile_options(${target} PRIVATE -UNDEBUG)
    endforeach()
    foreach(kind partition spmv cg)
        foreach(ranks 1 2 4)
            add_test(NAME mpi_${kind}_${ranks}
                COMMAND ${MPIEXEC_EXECUTABLE} ${MPIEXEC_NUMPROC_FLAG} ${ranks}
                    ${MPIEXEC_PREFLAGS} $<TARGET_FILE:test_mpi_${kind}> ${MPIEXEC_POSTFLAGS})
            set_tests_properties(mpi_${kind}_${ranks} PROPERTIES TIMEOUT 90 PROCESSORS ${ranks})
        endforeach()
    endforeach()
    add_test(NAME mpi_benchmark_empty_rank
        COMMAND ${MPIEXEC_EXECUTABLE} ${MPIEXEC_NUMPROC_FLAG} 4
            ${MPIEXEC_PREFLAGS} $<TARGET_FILE:test_mpi_benchmark> ${MPIEXEC_POSTFLAGS})
    set_tests_properties(mpi_benchmark_empty_rank PROPERTIES TIMEOUT 90 PROCESSORS 4)
    find_package(Python3 REQUIRED COMPONENTS Interpreter)
    set(launcher_flags)
    foreach(flag IN LISTS MPIEXEC_PREFLAGS)
        list(APPEND launcher_flags "--preflag=${flag}")
    endforeach()
    foreach(flag IN LISTS MPIEXEC_POSTFLAGS)
        list(APPEND launcher_flags "--postflag=${flag}")
    endforeach()
    add_test(NAME mpi_runner_safety
        COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/check_runner.py
            --launcher=${MPIEXEC_EXECUTABLE} --numproc-flag=${MPIEXEC_NUMPROC_FLAG}
            --runner=$<TARGET_FILE:run_mpi_benchmark> ${launcher_flags})
    set_tests_properties(mpi_runner_safety PROPERTIES TIMEOUT 360 PROCESSORS 2)
endif()
